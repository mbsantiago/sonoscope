<script>
  import { Sonoscope } from '@sonoscope/core';

  const audioUrl = '/audio/marico-sunbird.ogg';

  const scope = await Sonoscope.fromUrl(audioUrl);

  const topRulerCanvas = document.querySelector('#canvas-top-ruler');
  const time = scope.createTimeRuler(topRulerCanvas, { tickPosition: "bottom" });

  const leftRulerCanvas = document.querySelector('#canvas-freq-ruler');
  const freq = scope.createFrequencyRuler(leftRulerCanvas);

  const specCanvas = document.querySelector('#canvas-spectrogram');
  const spectrogram = scope.createSpectrogram(specCanvas);

  scope.attachNavigation(specCanvas, { axis: "both" });
</script>
