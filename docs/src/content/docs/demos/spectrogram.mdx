---
title: Spectrogram
description: Minimal standalone HTML & JavaScript spectrogram viewer.
---

Simple, zero-framework spectrogram rendering on a `<canvas>` element using `@sonoscope/core`.

```astro live
<div id="spectrogram-container">
  <canvas id="spectrogram-canvas" ></canvas>
</div>

<script>
  import { Sonoscope } from '@sonoscope/core';

  const canvas = document.querySelector('#spectrogram-canvas');
  const audioUrl = '/audio/marico-sunbird.ogg';

  const scope = await Sonoscope.fromUrl(audioUrl, {
    frequencyScale: 'mel',
  });

  const spec = scope.createSpectrogram(canvas, {
    colorMap: 'inferno',
    minValue: -80,
    maxValue: 0,
  });

  scope.attachNavigation(canvas);
</script>

<style>
  #spectrogram-container {
    border: 1px solid rgba(128, 128, 128, 0.25);
    overflow: hidden;
    background: rgba(0, 0, 0, 0.03);
  }

  #spectrogram-canvas {
    width: 100%;
    height: 280px;
    display: block;
  }
</style>

```
