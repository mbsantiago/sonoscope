---
title: React Components
description: Declarative audio visualization with @sonoscope/react components and hooks.
---

Declarative JSX composition using `@sonoscope/react`. Viewport synchronization, zoom/pan navigation, and audio playhead tracking are coordinated by `<SonoscopeProvider />`.

```tsx live props={{ "client:only": "react" }}
import {
  FrequencyRuler,
  SonoscopeProvider,
  Spectrogram,
  TimeRuler,
  Waveform,
} from "@sonoscope/react";
import { useRef, useState } from "react";

export default function ReactSpectrogramViewer() {
  const audioRef = useRef<HTMLAudioElement>(null);
  const audioUrl =
    "https://upload.wikimedia.org/wikipedia/commons/c/c5/Marico_Sunbird_%28Nectarinia_mariquensis%29_%28W1CDR0000941_BD17%29.ogg?utm_source=commons.wikimedia.org&utm_campaign=index&utm_content=original";

  return (
    <div className="not-content my-4 font-sans">
      <SonoscopeProvider
        url={audioUrl}
        frequencyScale="mel"
        followPlayback="page"
        audio={audioRef.current || undefined}
      >
        <div className="grid grid-cols-[56px_1fr] overflow-hidden rounded border border-[var(--sl-color-hairline-light,rgba(128,128,128,0.25))]">
          <div className="flex h-6 items-center justify-center border-r border-b border-[var(--sl-color-hairline,rgba(128,128,128,0.2))] font-mono text-[10px] opacity-60">
            Hz \ s
          </div>
          <div className="relative h-6 border-b border-[var(--sl-color-hairline,rgba(128,128,128,0.2))]">
            <TimeRuler
              height={24}
              tickPosition="top"
              color="rgba(128, 128, 128, 0.75)"
              tickColor="rgba(128, 128, 128, 0.35)"
            />
          </div>

          <div className="relative h-[260px] w-14 border-r border-[var(--sl-color-hairline,rgba(128,128,128,0.2))]">
            <FrequencyRuler
              width={56}
              tickPosition="right"
              color="rgba(128, 128, 128, 0.75)"
              tickColor="rgba(128, 128, 128, 0.35)"
            />
          </div>
          <div className="relative h-[260px]">
            <Spectrogram
              height={260}
              colorMap="plasma"
              minValue={-80}
              maxValue={0}
            />
          </div>

          <div className="flex h-[65px] items-center justify-center border-r border-t border-[var(--sl-color-hairline,rgba(128,128,128,0.2))] font-mono text-[10px] opacity-60">
            WAV
          </div>
          <div className="relative h-[65px] border-t border-[var(--sl-color-hairline,rgba(128,128,128,0.2))]">
            <Waveform height={65} colorMap="plasma" />
          </div>
        </div>
      </SonoscopeProvider>

      <audio
        ref={audioRef}
        src={audioUrl}
        crossOrigin="anonymous"
        controls
        className="mt-3 h-9 w-full outline-none"
      />
    </div>
  );
}
```
