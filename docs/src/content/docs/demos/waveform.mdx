```astro live
<div id="waveform-container">
  <canvas id="waveform-canvas" ></canvas>
</div>

<script>
  import { Sonoscope } from '@sonoscope/core';

  const canvas = document.querySelector('#waveform-canvas');
  const audioUrl = '/audio/marico-sunbird.ogg';

  const scope = await Sonoscope.fromUrl(audioUrl);
  const wave = scope.createWaveform(canvas);
  scope.attachNavigation(canvas);
</script>

<style>
  #waveform-container {
    border: 1px solid rgba(128, 128, 128, 0.25);
    overflow: hidden;
    background: rgba(0, 0, 0, 0.03);
  }

  #waveform-canvas {
    width: 100%;
    height: 280px;
    display: block;
  }
</style>
```
